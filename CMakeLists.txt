cmake_minimum_required(VERSION 3.22)
project(gunblade_vr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_subdirectory(vendor/soundtouch)
add_library(gunblade_audio SHARED native/audio_stretch.cpp)
option(GUNBLADE_AUDIO_DIAGNOSTICS "Enable one-use bounded audio/timing capture (also requires audio-capture.on)" OFF)
target_compile_definitions(gunblade_audio PRIVATE GUNBLADE_AUDIO_BUILD)
if(GUNBLADE_AUDIO_DIAGNOSTICS)
 target_compile_definitions(gunblade_audio PRIVATE GUNBLADE_AUDIO_DIAGNOSTICS=1)
endif()
target_link_libraries(gunblade_audio PRIVATE gunblade_soundtouch Threads::Threads)
set_target_properties(gunblade_audio PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
 set_target_properties(gunblade_audio PROPERTIES PREFIX "")
endif()
add_library(gunblade_xr SHARED native/xr_host.cpp native/gl_scene.cpp native/gl_context.cpp native/presentation.cpp
 native/shared/xr_panel.cpp native/shared/ui_compat.cpp native/shared/glb_mesh.cpp vendor/glad/src/gl.c)
target_include_directories(gunblade_xr PRIVATE native native/shared vendor vendor/stb vendor/glad/include vendor/glm vendor/openxr/include)
if(WIN32)
 set_target_properties(gunblade_xr PROPERTIES PREFIX "")
 target_link_libraries(gunblade_xr PRIVATE opengl32 gdi32 user32)
 target_compile_definitions(gunblade_xr PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

  add_executable(${test}_test tests/${test}_test.cpp)
  target_compile_options(${test}_test PRIVATE -UNDEBUG)
  add_test(NAME ${test} COMMAND ${test}_test)
 endforeach()
 add_executable(presentation_policy_test tests/presentation_policy_test.cpp)

  target_compile_definitions(audio_capture_test PRIVATE GUNBLADE_AUDIO_DIAGNOSTICS=1 GUNBLADE_AUDIO_CAPTURE_TEST=1)

 target_include_directories(model2_renderer_test PRIVATE native vendor/glm)

  native/shared/xr_panel.cpp native/shared/ui_compat.cpp native/shared/glb_mesh.cpp vendor/glad/src/gl.c)
 target_include_directories(presentation_test PRIVATE native native/shared vendor vendor/stb vendor/glad/include vendor/glm)

 foreach(menu_test menu_runtime settings_layout menu_capture controller_label_layout)
  add_executable(${menu_test}_test tests/${menu_test}_test.cpp native/presentation.cpp native/gl_context.cpp
   native/shared/xr_panel.cpp native/shared/ui_compat.cpp native/shared/glb_mesh.cpp vendor/glad/src/gl.c)
  target_include_directories(${menu_test}_test PRIVATE native native/shared vendor vendor/stb vendor/glad/include vendor/glm)
  target_compile_definitions(${menu_test}_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${menu_test}_test PRIVATE -UNDEBUG)
  target_link_libraries(${menu_test}_test PRIVATE opengl32 gdi32 user32)
  if(menu_test STREQUAL "settings_layout")
   target_compile_definitions(${menu_test}_test PRIVATE GUNBLADE_SETTINGS_RENDER_TEST)
   add_test(NAME ${menu_test} COMMAND ${menu_test}_test "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/settings-menu-captures")
  elseif(menu_test STREQUAL "controller_label_layout")
   add_test(NAME ${menu_test} COMMAND ${menu_test}_test "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/controller-label-captures")
  elseif(menu_test STREQUAL "menu_capture")
   add_test(NAME ${menu_test} COMMAND ${menu_test}_test "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/actual-menu-captures")
  else()
   add_test(NAME ${menu_test} COMMAND ${menu_test}_test "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  endif()
 endforeach()
else()
 target_link_libraries(gunblade_xr PRIVATE EGL GLESv3 android log dl)
endif()
